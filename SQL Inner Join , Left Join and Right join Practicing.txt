Create Database Zomato;
Use Zomato;
CREATE TABLE Items (

    Item_ID INT PRIMARY KEY,

    Item_Name VARCHAR(50),

    Category VARCHAR(30),

    Price DECIMAL(10,2)

);

CREATE TABLE Orders (

    Order_ID INT PRIMARY KEY,

    Customer_Name VARCHAR(50),

    Item_ID INT,

    Quantity INT,

    FOREIGN KEY (Item_ID) REFERENCES Items(Item_ID)

Select customer_name, Item_Name, Quantity from Items right join Orders
